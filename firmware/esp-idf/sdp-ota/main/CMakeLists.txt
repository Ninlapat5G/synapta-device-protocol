idf_component_register(SRCS "sdp_ota.c" INCLUDE_DIRS "."
    PRIV_REQUIRES app_update bootloader_support esp_driver_uart esp_event esp_http_server esp_netif esp_wifi json mbedtls nvs_flash)
foreach(name IN ITEMS SDP_PROJECT_ID SDP_BUILD_ID)
    if(NOT DEFINED ${name} OR NOT "${${name}}" MATCHES "^[a-z0-9][a-z0-9_-]*$")
        message(FATAL_ERROR "Set ${name} to a valid SynaptaCode identifier")
    endif()
    string(LENGTH "${${name}}" id_length)
    if(id_length GREATER 96 OR "${${name}}" MATCHES "^(con|prn|aux|nul|com[0-9]|lpt[0-9])$")
        message(FATAL_ERROR "Invalid SynaptaCode identifier ${name}")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE "${name}=\"${${name}}\"")
endforeach()
if(SDP_OTA_TEST_REJECT)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE SDP_OTA_TEST_REJECT=1)
endif()
target_link_libraries(${COMPONENT_LIB} INTERFACE "-u sdp_app_desc")
